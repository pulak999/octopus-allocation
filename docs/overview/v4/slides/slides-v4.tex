% SLIDE 1 — Three takeaways upfront
% =========================================================================

\begin{frame}{Three takeaways.}

  \begin{enumerate}
    \setlength{\itemsep}{1.0em}

    \item \textbf{RL beats greedy dramatically on two held-out traces.}
          The learned policy achieves near-break-even pooling savings while
          greedy needs 4--13$\times$ more MPD capacity.
          The system generalises to unseen workloads.

    \item \textbf{The training system is not the bottleneck.}
          Data augmentation across 10 traces, a redesigned reward, and
          32 parallel environments eliminated the collapse failure and enabled
          generalisation.

    \item \textbf{The remaining ceiling is a reward design problem.}
          % VET: This is interpretive. Confirm before presenting.
          Savings plateau at $\sim$1--2\% from 100k steps onward with no
          improvement over 2M steps.
          The agent learns to game the proxy reward rather than maximise
          true pooling savings.
  \end{enumerate}

\end{frame}

% SLIDE 4 — Training system: what changed
% =========================================================================

\begin{frame}{Two independent improvements were needed to get here.}

  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \begin{block}{1. Data: augmentation across 10 traces}
        \small
        \begin{itemize}
          \setlength{\itemsep}{0.25em}
          \item \textbf{Before:} trained on one trace, one pod mapping.
                Policy memorised a single datacenter's pattern.
          \item \textbf{After:} 7 training traces, random pod seed
                per episode, memory scale jitter $s \in [0.7, 1.2]$.
          \item Episodes are diverse in workload intensity and
                VM arrival patterns.
        \end{itemize}
      \end{block}

      \vspace{0.5em}

      \begin{block}{2. Throughput: 32 parallel environments}
        \small
        \begin{itemize}
          \setlength{\itemsep}{0.25em}
          \item Bottleneck was CPU serial rollout, not GPU.
          \item \texttt{SubprocVecEnv(N=32)}: \textbf{26$\times$} speedup
                ($46 \to 1{,}221$ steps/sec).
          \item 2M steps now takes $\sim$14 min of training time.
        \end{itemize}
      \end{block}

% SLIDE 5 — Reward redesign: departure relief
% =========================================================================

\begin{frame}{The reward was redesigned to score durable load, not raw load.}

  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \begin{block}{Old reward (run 6 --- collapsed)}
        \small
        Penalise increase in the \emph{single worst MPD's} raw load.
        \begin{itemize}
          \setlength{\itemsep}{0.2em}
          \item All other MPDs invisible --- zero gradient signal.
          \item No departure model: 80\,GB departing in 10 minutes
                looks the same as 80\,GB staying for 3 days.
          \item \textbf{Result:} entropy collapsed at 100k steps;
                policy jammed all load onto one MPD.
        \end{itemize}
      \end{block}

    \column{0.50\textwidth}
      \begin{block}{New reward A (run 7 --- stable)}
        \small
        Penalise the worst \emph{projected sticky load} among
        all MPDs this host can reach.
        \begin{itemize}
          \setlength{\itemsep}{0.2em}
          \item Sticky load = current load $-$ memory leaving in $\sim$17\,hrs.
          \item Every reachable MPD generates signal every step.
          \item \textbf{Result:} entropy healthy throughout 2M steps;
                load spread evenly.
        \end{itemize}
      \end{block}
  \end{columns}

  \vspace{0.5em}

  \begin{center}

% =========================================================================
% SLIDE 7 — What the agent actually learned (reward hacking)
% =========================================================================

\begin{frame}{The agent is gaming the proxy reward, not maximising savings.}

  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \begin{block}{What we asked for}
        \small
        Minimise the worst \emph{projected sticky load} among reachable MPDs.
        We hoped this would translate to low peak load over the full trace.
      \end{block}

      \vspace{0.3em}

      \begin{block}{What the agent learned}
        \small
        Spread memory uniformly across reachable MPDs at every step.
        This minimises the per-step proxy reward perfectly ---
        but it \emph{ignores topology}: not all MPDs are equally
        shared, so uniform spread is not the same as globally balanced
        allocation.
      \end{block}

      \vspace{0.3em}

      \begin{block}{The gap}
        \small
        % VET: This is interpretive. Confirm with ablation if possible.
        The proxy reward (one step, one host's MPDs) cannot see the
        downstream effect of placing long-lived VMs on MPDs that
        many future hosts will also need.
      \end{block}

% SLIDE 8 — Next steps

\begin{frame}{Two concrete fixes for tomorrow.}

  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \begin{block}{Fix 1: Optimal-gap reward shaping}
        \small
        We have an exact solver (\texttt{find\_optimal}) that computes
        the best achievable peak for any snapshot of host demands
        using max-flow.

        \vspace{0.4em}
        Add a penalty proportional to how far above optimal the
        agent's allocation sits:
        \[
          r_t' = r_t^A - \beta \cdot
          \frac{c_{\max}^{\text{post}} - t^*(s_t)}{D_{\text{pod}}}
        \]

        \vspace{0.2em}
        The agent now knows exactly how far from optimal it is at
        every step --- not just whether the proxy improved.
      \end{block}

    \column{0.50\textwidth}
      \begin{block}{Fix 2: Behavioral cloning warm-start}
        \small
        The solver also produces optimal allocation \emph{weights}
        (flow solution $\to$ GB per MPD $\to$ normalised proportions).

        \vspace{0.4em}
        Pre-train the actor on these $(s, a^*)$ pairs before SAC
        starts.  The policy begins near ``instantaneously optimal''
        rather than random, preventing early entropy collapse.

        \vspace{0.4em}
        SAC then fine-tunes from that warm start toward
        temporally optimal (accounts for VM lifetimes).
      \end{block}
  \end{columns}
