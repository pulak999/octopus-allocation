\begin{frame}{Environment throughput improved by 2.73$\times$ after SPEEDUP\_PLAN.}

        \bottomrule
      \end{tabular}}

      \vspace{0.7em}
      \begin{itemize}
        \setlength{\itemsep}{0.55em}
        \item Chunks 1--3: flat arrays, disk-backed precompute cache, Numba kernels.
        \item Throughput gain vs.\ pre-SPEEDUP: \textbf{2.73$\times$}.
        \item Post-speedup steady windows stay in a tight 2,533--2,959 fps band.
        \item Same benchmark setup: 300s env-only run, 32 envs, reward variant A.
      \end{itemize}

    \column{0.40\textwidth}
      \vspace{0.6em}
      \begin{block}{Cache behavior}

        First run cache prep: 379.5s\\
        Subsequent runs: \(\sim\)3s
      \end{block}

      \vspace{0.45em}
      \begin{block}{Interpretation}
        \small
        Environment stepping is no longer the dominant bottleneck in training.
      \end{block}
  \end{columns}

\end{frame}
